import InteractiveCode from '../../../components/InteractiveCode.astro';

<InteractiveCode
  css={`/* Static — default positioning (flows in normal document order) */
.static-box {
  position: static;
}

/* Relative — offset from its normal position */
.relative-box {
  position: relative;
  top: 20px;
  left: 10px;
}

/* Absolute — positioned relative to nearest positioned ancestor */
.absolute-box {
  position: absolute;
  top: 0;
  right: 0;
}

/* Fixed — positioned relative to the viewport, stays on scroll */
.fixed-box {
  position: fixed;
  bottom: 0;
  width: 100%;
}`}
/>
